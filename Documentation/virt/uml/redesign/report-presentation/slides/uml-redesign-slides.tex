\begin{frame}{What KVM v1 Taught Us}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{v1 design pressure}
\begin{itemize}
  \item treat KVM as a faster trap loop
  \item share too much vCPU/run state
  \item repair state after exits
  \item maintain shadow page-table state
\end{itemize}
\end{block}
\column{0.48\textwidth}
\begin{block}{lesson}
\begin{itemize}
  \item SMP turns repair workarounds into races
  \item \texttt{kvm\_run} must have one active owner
  \item host signals need KVM-visible masking
  \item every state item needs an authoritative home
\end{itemize}
\end{block}
\end{columns}
\vspace{0.35em}
\centering
\pill{UMLBlue}{DESIGN SHIFT}\quad from ``fix the next trap'' to an explicit state-ownership contract
\end{frame}

\begin{frame}{Changed Data Model}
\scriptsize
\begin{tabularx}{\textwidth}{@{}p{0.25\textwidth}p{0.25\textwidth}X@{}}
\toprule
State class & Authoritative home & Contract in v2 \\
\midrule
\texttt{kvm\_run} and vCPU registers & per-host-CPU vCPU & Pick while migration is disabled; no other CPU owns that run page during dispatch. \\
Guest GPRs and syscall ABI & task registers plus sync-regs mmap & Marshal into KVM before entry and back to \UML{} after exit at explicit boundaries. \\
FPU/XSAVE & per-task snapshots plus vCPU dirty epoch & Install on first use, capture when ownership changes, avoid cross-task XMM leakage. \\
IST and exception frames & per-task frame snapshots over per-vCPU pages & Restore before dispatch; snapshot interrupted stubs instead of reading another task's frame. \\
TLB state & per-mm generation plus per-vCPU last-seen state & Memory changes become generation changes; vCPUs refresh or get kicked. \\
Host signal delivery & \texttt{KVM\_SET\_SIGNAL\_MASK} & \texttt{SIGALRM} cannot arbitrarily break the KVM window; a dedicated kick signal can. \\
\bottomrule
\end{tabularx}
\end{frame}

\begin{frame}{The R14 Cache-Flake Saga in One Slide}
\scriptsize
\begin{columns}[T,onlytextwidth]
\column{0.49\textwidth}
\begin{block}{Five weeks, fourteen rounds}
\begin{itemize}

  \item KVMvTwo{} only; seccomp clean
  \item Eliminated through R1--R12: KSM, khugepaged, kswapd, host pagecache, fast-page-fault, TLB coherence, mmu\_notifier, UML self-unmap, physmem residency, CPython specialization, gadget bounds (T68 shipped as side-effect)
  \item R13 ruled out MAP\_POPULATE pinning as the cause (kept as foundation T71)
\end{itemize}
\end{block}
\column{0.49\textwidth}
\begin{block}{Round 14: root cause + fix}
\begin{itemize}
  \item Corruption shape: 16 valid bytes + 16 zero bytes = half AVX2 vector
  \item SMP-T57 had un-masked AVX in CPUID; XCR0=0x7
  \item glibc IFUNC picked AVX2 \texttt{vmovdqu ymm} memcpy
  \item Per-task FPU save/restore used legacy \texttt{KVM\_GET\_FPU} (512~B FXSAVE)
  \item \emph{No YMM upper-128 coverage} $\Rightarrow$ cross-task leak
  \item Fix (SMP-T73): \texttt{KVM\_GET\_FPU} $\to$ \texttt{KVM\_GET\_XSAVE} at 5 sites
  \item 220/220 validation soak; $p = 0.007$
\end{itemize}
\end{block}
