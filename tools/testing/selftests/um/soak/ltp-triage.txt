# These suites are not in the default kirk invocation. They require a
# one-shot verification cycle to
# confirm they CONF-out cleanly under UML rather than FAIL; once
# verified, promote to ltp-suite.txt and drop the entry here.
#
# Verification pattern:
#   python3 /opt/kirk/kirk --sut host \
#       --run-suite <one-of-these> \
#       --skip-file ltp-skip.txt \
#       --json-report /tmp/triage-<name>.json
#   # then inspect FAIL/CONF/PASS counts in the JSON.

# UML has no NUMA topology; expect CONF, not FAIL.
numa

# AF_ALG socket tests; CONFIG_CRYPTO_USER_API=y is in the
# UML reference build, so these should pass.
crypto

# growfiles/fsstress on TMPDIR; on UML the TMPDIR is
# hostfs-backed, so "filesystem stress" becomes "hostfs stress on
# the host's real FS". Useful as a hostfs torture test if operator
# accepts that semantics.
fs

# DAC permission tests; depends on hostfs honouring
# the expected chown/chmod semantics.
fs_perms_simple

# POSIX fcntl(F_SETLK) tests; UML's range-lock implementation needs
# explicit verification.
fcntl-locktests

# wqueue01-09; depends on CONFIG_WATCH_QUEUE; if the
# UML reference config doesn't enable it, tests CONF-out cleanly.
watchqueue

# Tests of host userspace commands (cp, mv, ln, ar,
# tar, gzip). Useful smoke of the whole guest userspace; uncertain
# whether the test rig assumes a richer shell environment than
# the UML guest provides.
commands

# POSIX conformance; partly overlaps `syscalls`. Defer until the core
# syscall suite is green.
open_posix_testsuite
